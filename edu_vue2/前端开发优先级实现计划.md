## 关键实现建议

### 基于课程的权限系统实现
```javascript
// store/modules/coursePermission.js - 新建
// 用户在各课程中的权限
export const state = {
  // userCourses[userId] = [
  //   { courseId: 1, role: 'creator' },
  //   { courseId: 2, role: 'student' },
  //   { courseId: 3, role: 'student' }
  // ]
  userCourses: {},
  myCreatedCourses: []  // 用户创建的课程列表
}

export const mutations = {
  SET_USER_COURSES(state, courses) {
    state.userCourses = courses
  },
  SET_MY_CREATED_COURSES(state, courses) {
    state.myCreatedCourses = courses
  }
}

export const getters = {
  // 检查用户是否为某课程的创建者
  isCourseCreator: (state) => (courseId) => {
    return state.myCreatedCourses.some(c => c.id === courseId)
  },
  // 获取用户在某课程中的角色
  getUserRoleInCourse: (state) => (courseId) => {
    const course = state.userCourses[courseId]
    return course ? course.role : null
  }
}
```
